Cấu hình Supabase & SQL Schema

Bạn có thể nhập Project URL và Anon Key của dự án Supabase để đồng bộ dữ liệu thật:

SQL Script Khởi tạo (Copy dán vào Supabase SQL Editor) 
-- 1. Bảng travel_trips
create table if not exists public.travel_trips (
  id uuid default gen_random_uuid() primary key,
  title text not null,
  location text not null,
  lat float8 not null,
  lng float8 not null,
  distance_km int4 default 0,
  travel_date date not null,
  vehicle text not null,
  description text,
  image_urls text[] default array[]::text[],
  created_at timestamp with time zone default timezone('utc'::text, now()) not null
);

-- 2. Kích hoạt Row Level Security (RLS)
alter table public.travel_trips enable row level security;

-- 3. Cho phép công chúng đọc (anonymous SELECT)
create policy "Allow public read trips"
  on public.travel_trips for select
  using (true);

-- 4. Cho phép người dùng đã xác thực Thêm/Sửa/Xóa
create policy "Allow auth insert trips"
  on public.travel_trips for insert
  to authenticated
  with check (true);

create policy "Allow auth update trips"
  on public.travel_trips for update
  to authenticated
  using (true);

create policy "Allow auth delete trips"
  on public.travel_trips for delete
  to authenticated
  using (true);

-- 5. Bucket 'travel-photos' cho Supabase Storage
insert into storage.buckets (id, name, public)
values ('travel-photos', 'travel-photos', true)
on conflict (id) do nothing;

create policy "Allow public view photos"
  on storage.objects for select
  using (bucket_id = 'travel-photos');

create policy "Allow auth upload photos"
  on storage.objects for insert
  to authenticated
  with check (bucket_id = 'travel-photos');
